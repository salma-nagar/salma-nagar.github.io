function showTab(tabId) { // إخفاء كل المحتويات أولاً document.querySelectorAll('.tab-content').forEach(content => { content.classList.remove('active'); }); // إظهار التبويب المختار const targetTab = document.getElementById(tabId); if(targetTab) targetTab.classList.add('active'); // تحديث مظهر الروابط النشطة في ديسكتوب document.querySelectorAll('.nav-link').forEach(link => { link.classList.remove('text-[#F59E0B]', 'font-bold', 'border-b-2', 'border-[#F59E0B]'); link.classList.add('text-stone-400'); }); // تحديث مظهر الأزرار النشطة في الموبايل document.querySelectorAll('.mobile-nav-link').forEach(btn => { btn.classList.remove('bg-[#F59E0B]', 'text-black', 'font-bold'); btn.classList.add('text-stone-400', 'bg-stone-900/40'); }); } function scrollToSection(id) { const el = document.getElementById(id); if (el) el.scrollIntoView({ behavior: 'smooth' }); }
